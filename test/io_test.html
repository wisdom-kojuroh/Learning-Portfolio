// js/io.js function exportJSON(flashcards) { const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(flashcards, null, 2)); const downloadAnchor = document.createElement('a'); downloadAnchor.setAttribute("href", dataStr); downloadAnchor.setAttribute("download", "flashcards_backup.json"); document.body.appendChild(downloadAnchor); downloadAnchor.click(); downloadAnchor.remove(); } function importJSON(file, callback) { const reader = new FileReader(); reader.onload = function(e) { try { const parsed = JSON.parse(e.target.result); const normalized = normalizeCardList(parsed); if (normalized) { callback(null, normalized); } else { callback(new Error("有効なカードデータが見つからないよ。")); } } catch (err) { callback(err); } }; reader.readAsText(file); }
